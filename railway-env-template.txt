# ─────────────────────────────────────────────────────────────────────────────────────
# DATABASE CONNECTION (PostgreSQL)
# ─────────────────────────────────────────────────────────────────────────────────────
# Standard connection string from Railway PostgreSQL service
# Format: postgresql://USER:PASSWORD@HOST:PORT/DATABASE?schema=public&sslmode=require
# Railway provides this automatically when you link a Postgres service.
# Replace with actual Railway Postgres URL after creating the service.
DATABASE_URL=postgresql://USER:PASSWORD@HOST:PORT/DATABASE?schema=public&sslmode=require

# Direct database connection (for Prisma migrations)
# Use native PostgreSQL (not PgBouncer) to avoid pooling issues with migrations
# When using connection pooling (PgBouncer), set this to a direct connection URL
DIRECT_URL=postgresql://USER:PASSWORD@HOST:PORT/DATABASE?schema=public&sslmode=require

# ─────────────────────────────────────────────────────────────────────────────────────
# AUTHENTICATION & SECURITY SECRETS
# ─────────────────────────────────────────────────────────────────────────────────────
# CRITICAL: All secrets below MUST be strong, random, and unique.
# Minimum 32 characters for all secret values.
# Generated example: node -e "console.log(require('crypto').randomBytes(32).toString('base64'))"

# NextAuth.js Secret (generate new for production)
# Used by NextAuth for session encryption and CSRF protection
NEXTAUTH_SECRET=GENERATE_UNIQUE_SECRET_MIN_32_CHARS

# JWT Access Token Secret (generate new for production)
# Used to sign short-lived access tokens for admin authentication (15m default)
# If compromised, access is limited to 15 minutes
JWT_ACCESS_SECRET=GENERATE_UNIQUE_SECRET_MIN_32_CHARS

# JWT Refresh Token Secret (generate new for production)
# Used to sign long-lived refresh tokens (7 days default)
# Stored securely on client and database
JWT_REFRESH_SECRET=GENERATE_UNIQUE_SECRET_MIN_32_CHARS

# JWT Claims (optional, for token metadata)
# Identifies your service in JWT tokens
JWT_ISSUER=capital-network-api
JWT_AUDIENCE=capital-network-web

# ─────────────────────────────────────────────────────────────────────────────────────
# ADMIN ACCOUNT (used only by Prisma seed on first deployment)
# ─────────────────────────────────────────────────────────────────────────────────────
# These are ONLY used during: npm run seed or prisma db seed
# After first deployment, change password via admin panel.
# NEVER reuse these values across environments.

# Admin email (used for login)
ADMIN_EMAIL=admin@yourdomain.com

# Admin username (fallback, usually same as email)
ADMIN_USERNAME=admin

# Admin initial password (minimum 8 characters)
# CRITICAL: Must be changed immediately after first login in production
# Generate: openssl rand -base64 12
# DO NOT reuse passwords across environments (dev, staging, production)
ADMIN_PASSWORD=GENERATE_STRONG_PASSWORD_MIN_8_CHARS

# Admin password salt (used for password hashing)
# Generate: node -e "console.log(require('crypto').randomBytes(32).toString('base64'))"
# Must be consistent for password verification across restarts
ADMIN_PASSWORD_SALT=GENERATE_UNIQUE_SALT_MIN_32_CHARS
